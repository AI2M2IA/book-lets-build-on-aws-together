% Fix pandoc syntax-highlighted code blocks (Shaded/Highlighting):
%   - fvextra enables breaklines inside fancyvrb Verbatim environments
%   - Redefine Highlighting with breaklines so long code lines wrap instead of overflow
%   - Redefine Shaded with framed's shaded* (breakable across pages) instead of snugshade
\usepackage{fvextra}
\usepackage{framed}
\AtBeginDocument{%
  \DefineVerbatimEnvironment{Highlighting}{Verbatim}{%
    breaklines=true,%
    breaksymbolleft={},%
    breakanywhere=true,%
    commandchars=\\\{\},%
    fontsize=\small%
  }%
  \colorlet{shadecolor}{nimbuscodebg}%
  \renewenvironment{Shaded}{\begin{shaded*}}{\end{shaded*}}%
}
